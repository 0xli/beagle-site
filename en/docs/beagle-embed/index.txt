function under(root, rel) {
  const base = normalize(root + "/");
  const file = normalize(join(root, rel));
  return file.startsWith(base) ? file : null;
}

function fileFor(url) {
  const path = decodeURIComponent((url || "/").split("?")[0]);
  if (path === "/" || path === "/example" || path === "/example/") return join(pkg, "example/index.html");
  if (path === "/beagle-frame.html") return join(pkg, "frame/beagle-frame.html");
  if (path.startsWith("/example/") || path.startsWith("/src/")) return under(pkg, path.slice(1));
  if (path === "/vendor/nacl.js") return nacl;
  if (path === "/vendor/beagle-ui.js") return ui;
  if (path === "/beagle-web.js" || path.startsWith("/vendor/")) return under(web, path.slice(1));
  return null;
}

if (!web) throw new Error("set BEAGLE_WEB_PUBLIC to a beagle-web public/ directory");

const port = Number(process.env.PORT || 0);
const server = createServer((req, res) => {
  const file = fileFor(req.url);
  if (!file || !existsSync(file) || !statSync(file).isFile()) {
    res.writeHead(404);
    res.end("not found");
    return;
  }
  res.writeHead(200, {
    "content-type": types[extname(file)] || "application/octet-stream",
    "cache-control": "no-store",
  });
  createReadStream(file).pipe(res);
});
