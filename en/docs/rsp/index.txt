// Two throwaway identities in one process, over an RSP/1 node that you run.
import { generateKeyPairSync, randomUUID } from "node:crypto";
import { RendezvousClient, carrierSigner, routeCall } from "@decentnetwork/rendezvous-client";
import { base58ToBytes, bytesToBase58, createCarrierRendezvousAuthProof } from "@decentnetwork/peer";

const NODE = process.env.RSP_NODE_URL;
if (!NODE) throw new Error("set RSP_NODE_URL to an RSP/1 node that you run");

function identity() {
  const jwk = generateKeyPairSync("x25519").privateKey.export({ format: "jwk" });
  const secret = new Uint8Array(Buffer.from(jwk.d, "base64url"));
  const userid = bytesToBase58(new Uint8Array(Buffer.from(jwk.x, "base64url")));
  const signer = carrierSigner({ userid, createProof: ({ serverPublicKey, transcript }) =>
    createCarrierRendezvousAuthProof(secret, base58ToBytes(serverPublicKey), transcript) });
  return { id: { method: "carrier-x25519", id: userid }, signer };
}

// A friend's `extra` is not the shape routeCall takes: convert it.
function advertisement(extra) {
  const rsp = JSON.parse(extra || "{}").rsp;
  if (rsp?.protocol === "rsp/1" && rsp.nodeUrl === NODE) return { rsp: true, nodeUrl: NODE };
}

const alice = identity();
const bob = identity();

// Bob listens, and rings only for a friend.
const listener = new RendezvousClient(NODE, bob.signer);
let invite;
listener.on("call:invite", async (i) => {
  if (i.caller.id !== alice.id.id) return; // not a friend: stay silent
  invite = { callId: i.callId, peer: i.caller, callExpiresAt: i.expiresAt, capability: i.joinCapability };
  await listener.ringing(invite);
});
await listener.listen();

// Alice calls. Carrier first, then RSP, then socket.io.
const bobExtra = JSON.stringify({ rsp: { protocol: "rsp/1", nodeUrl: NODE } });
const call = await routeCall({
  tryCarrier: async () => false, // pretend Bob is not reachable over Carrier
  advertisement: advertisement(bobExtra),
  signer: alice.signer,
  callId: randomUUID(),
  peer: bob.id,
  push: async () => { throw new Error("nobody rang: send a sealed wake push here"); },
  socketFallback: async () => ({ path: "socket.io" }),
});
console.log("path:", call.path);

// Bob answers. WebRTC offer and answer would follow on the same clients.
if (call.client) {
  const accepted = new Promise((resolve) => call.client.on("signal", resolve));
  await listener.join(invite);
  await listener.sendSignal({
    version: 1, callId: invite.callId, type: "call.accept", sequence: 1,
    timestamp: Date.now(), expiresAt: invite.callExpiresAt, encoding: "json", payload: "{}",
  });
  console.log("received:", (await accepted).type);
  call.client.close();
}
