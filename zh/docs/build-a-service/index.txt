function takeNonce(nonce) {
  const exp = nonces.get(nonce);
  nonces.delete(nonce);
  return exp !== undefined && exp >= Date.now();
}

// ── the one primitive: XEdDSA over the Beagle key, NOT Ed25519 ──
function verifyBeagle(prefix, userid, origin, salt, sigHex) {
  try {
    const msg = new TextEncoder().encode(`${prefix}\n${origin}\n${salt}`);
    return verifyDetached(base58ToBytes(userid), msg, Uint8Array.from(Buffer.from(sigHex, "hex")));
  } catch {
    return false;
  }
}

// ── session: an HMAC'd cookie keyed by the Beagle userid ────────
const mac = (v) => createHmac("sha256", SECRET).update(v).digest("hex");

function setSession(res, userid) {
  res.setHeader("Set-Cookie",
    `sid=${encodeURIComponent(`${userid}.${mac(userid)}`)}; HttpOnly; Secure; SameSite=Lax; Path=/; Max-Age=2592000`);
}

function sessionUser(req) {
  const m = /(?:^|;\s*)sid=([^;]+)/.exec(req.headers.cookie || "");
  if (!m) return null;
  const v = decodeURIComponent(m[1]);
  const i = v.lastIndexOf(".");
  const userid = v.slice(0, i);
  const got = Buffer.from(v.slice(i + 1));
  const want = Buffer.from(mac(userid));
  return got.length === want.length && timingSafeEqual(got, want) ? userid : null;
}
